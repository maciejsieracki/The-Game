const path = require('path');
const GRA = '/home/user/The-Game/.claude/worktrees/wf_16ad639d-ca0-3/gra';

  const page = await browser.newPage({ viewport: { width: 1500, height: 950 } });
  const errs = []; page.on('pageerror', (e) => errs.push(String(e)));
  const dialogs = [];
  page.on('dialog', async (d) => { dialogs.push(d.message()); await d.accept(); });

    const left = row.querySelector('.arr');

  results.before = await readHud();

  await page.locator('[data-work-tax-submit]').click();

  console.log('PO', JSON.stringify(results.after));

  const clicked = await page.evaluate(() => { const el = Array.from(document.querySelectorAll('*')).filter((e) => /ZAKOŃCZ TURĘ|Zakończ turę/i.test(e.textContent || '') && (e.textContent || '').length < 40).pop(); if (!el) return false; el.click(); return true; });
