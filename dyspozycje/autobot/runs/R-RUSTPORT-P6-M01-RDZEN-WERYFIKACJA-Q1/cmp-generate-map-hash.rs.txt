use civ_engine::domain::map_generator::{generate_map, WorldType};

fn fnv(bytes: &[u8]) -> u64 {
    let mut h: u64 = 0xcbf29ce484222325;
    for b in bytes {
        h ^= u64::from(*b);
        h = h.wrapping_mul(0x100000001b3);
    }
    h
}

#[test]
fn print_hashes() {
    let mut differing_inputs = 0;
    let mut out = String::new();
    for (w, h) in [(36_u32, 28_u32), (24, 18), (60, 45), (168, 120), (5, 5), (7, 9), (3, 3), (40, 3), (100, 33), (12, 7)] {
        for seed in [42_u32, 7, 123, 99999] {
            for wt in [WorldType::Continents, WorldType::Pangea, WorldType::Islands, WorldType::Earth] {
                let result = std::panic::catch_unwind(|| generate_map(w, h, seed, wt));
                match result {
                    Ok(Ok(map)) => {
                        let json = serde_json::to_string(&map).expect("json");
                        out.push_str(&format!("{w}x{h} s{seed} {wt} {:016x}\n", fnv(json.as_bytes())));
                    }
                    Ok(Err(e)) => out.push_str(&format!("{w}x{h} s{seed} {wt} ERR {e}\n")),
                    Err(_) => out.push_str(&format!("{w}x{h} s{seed} {wt} PANIC\n")),
                }
                differing_inputs += 1;
            }
        }
    }
    std::fs::write(std::env::var("HASH_OUT").expect("HASH_OUT"), out).expect("write");
    assert!(differing_inputs > 0);
}
